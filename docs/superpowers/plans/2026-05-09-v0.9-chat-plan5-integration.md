- **Test file mirror layout:** `backend/tests/e2e/test_<feature>_cassette.py`,cassette yaml 自动落 `backend/tests/fixtures/cassettes/test_<feature>_cassette/<test_name>.yaml`

- **Run all commands from `backend/`** unless noted otherwise

"""L2 cassette 1 — v0.9 chat 基础 flow: ICBC 现价 → MCP get_stock_quote → SSE token stream → done.

Industry-level coverage (spec § 5):

import os

    """FastAPI app with PG checkpointer + MCP server lifecycle."""

@pytest.mark.vcr
def test_basic_icbc_quote_end_to_end(chat_app: TestClient) -> None:
    """User asks ICBC price → tool dispatch → response with price.

    Verifies SSE event sequence: token (planner thinking) → plan → tool_start →
    tool_end → token (responder) ... → done. Each event has monotonically
    increasing seq.
    """
    # 1. Create chat session
    r = chat_app.post("/api/v0/chats/", json={"title": "cassette basic"})
    assert r.status_code == 201
    sid = r.json()["id"]

    # 2. POST chat message and consume SSE
    events: list[dict] = []
    with chat_app.stream(
        "POST", "/api/v0/chat",
        json={"session_id": sid, "message": "工商银行现价多少"},
        timeout=60.0,
    ) as resp:
        assert resp.status_code == 200
        for line in resp.iter_lines():
            if not line or not line.startswith("data:"):
                continue
            payload = json.loads(line[len("data:"):].strip())
            events.append(payload)

    # 3. Assertions per industry problems
    event_types = [e["event"] for e in events]
    assert "plan" in event_types, "A1: planner must emit plan"
    assert "tool_start" in event_types, "M1: MCP tool must dispatch"
    assert "tool_end" in event_types, "M2: MCP tool must return"
    assert "done" in event_types, "G1: stream must close cleanly"

    # A1: planner selected get_stock_quote (not get_news / web_search)
    plan_event = next(e for e in events if e["event"] == "plan")
    assert any(tc["tool_name"] == "get_stock_quote" for tc in plan_event["data"]["tool_calls"]), (
        "A1: must select get_stock_quote for price query"
    )

    # A3: ts_code resolved correctly
    tool_start = next(e for e in events if e["event"] == "tool_start" and
                      e["data"]["tool_name"] == "get_stock_quote")
    assert tool_start["data"]["args"]["ts_code"] == "601398.SH", "A3: ts_code arg fidelity"

    # A4: no hallucinated tool names
    all_tool_names = {e["data"]["tool_name"] for e in events
                       if e["event"] in ("tool_start", "tool_end")}
    valid_tools = {"get_stock_quote", "get_financials", "get_news", "web_search",
                   "kb_search", "compare_stocks"}
    assert all_tool_names <= valid_tools, f"A4: hallucinated tool: {all_tool_names - valid_tools}"

    # G1: seq monotonic
    seqs = [e["seq"] for e in events if "seq" in e]
    assert seqs == sorted(seqs), "G1: SSE seq must be monotonic"

    # Final response must include a price-like signal
    done_event = next(e for e in events if e["event"] == "done")
    final_text = done_event["data"]["final_response"]
    assert "工商银行" in final_text or "601398" in final_text
    # Loose price check — model may say "约 X.XX 元" or similar
    import re
    assert re.search(r"\d+\.\d+", final_text), "response should mention numeric price"
```

"""L2 cassette 2 — multi-turn + tool dedup.

User asks the *same* tool args 3 turns in a row. Expect:
- Turn 1: tool_start + tool_end (cache MISS)
- Turn 2: cache HIT, no tool_start (within TTL window)
- Turn 3: cache HIT, no tool_start

Industry-level coverage:
- B3 tool result staleness (TTL respected)
- C1 multi-turn tool coordination (planner reuses cached result)
"""

from __future__ import annotations

import json
import os

import pytest
from fastapi.testclient import TestClient


@pytest.fixture
def vcr_config() -> dict:
    return {
        "filter_headers": ["authorization", "x-api-key", "cookie"],
        "record_mode": os.environ.get("VCR_RECORD_MODE", "none"),
        "match_on": ["method", "scheme", "host", "port", "path", "query"],
    }


@pytest.fixture
def chat_app(pg_test_container, monkeypatch):
    monkeypatch.setenv("DATABASE_URL", pg_test_container.dsn)
    from app.app_main import app
    with TestClient(app) as client:
        yield client


def _drain_sse(resp) -> list[dict]:
    out = []
    for line in resp.iter_lines():
        if line and line.startswith("data:"):
            out.append(json.loads(line[len("data:"):].strip()))
    return out


@pytest.mark.vcr
def test_multi_turn_tool_dedup(chat_app: TestClient) -> None:
    sid = chat_app.post("/api/v0/chats/", json={"title": "dedup"}).json()["id"]

    tool_starts_per_turn = []
    for turn in range(3):
        with chat_app.stream(
            "POST", "/api/v0/chat",
            json={"session_id": sid, "message": "工商银行现价多少"},
            timeout=60.0,
        ) as resp:
            events = _drain_sse(resp)
        tool_starts = [e for e in events if e["event"] == "tool_start" and
                        e["data"]["tool_name"] == "get_stock_quote"]
        tool_starts_per_turn.append(len(tool_starts))

    # Turn 1: 1 tool_start (MISS)
    # Turn 2 + 3: 0 tool_start (HIT, dedup)
    assert tool_starts_per_turn[0] >= 1, "B3: turn 1 should fetch tool"
    # C1: subsequent turns should NOT re-fetch (cache hit)
    # Allow planner to occasionally still emit if context changed; assert
    # at least 1 turn skipped fetch
    assert any(c == 0 for c in tool_starts_per_turn[1:]), (
        "C1: at least one of turn 2/3 should hit cache (no tool_start)"
    )
```

- [ ] **Step 1: Write extraction quality eval pipeline**

Create `backend/tests/eval/extraction_quality_eval.py`:

```python
"""E9 — LLM extraction quality eval pipeline for EscalationPacket.

Ground truth source: EscalationRecord.user_edits (PG jsonb field, populated
when user clicks "edit" on a draft field in the EscalationConfirmDialog).

Metrics:
- field_accuracy = 1 - (n_fields_user_edited / n_fields_total)
- entity_recall = (n_entities_LLM_caught_correctly / n_entities_user_kept)
- entity_precision = (n_entities_LLM_caught_correctly / n_entities_LLM_extracted)
- preference_F1 = harmonic mean of preference recall+precision
- missing_field_quality = 1 if user filled the field LLM marked missing,
  else 0 (means LLM correctly identified gap)

Output: JSON report aggregating metrics across all EscalationRecord rows.
Used in: CI baseline regression check + Plan 5 final acceptance.
"""

from __future__ import annotations

import asyncio
import json
import os
from dataclasses import dataclass, field
from typing import Any

import asyncpg


@dataclass
class ExtractionQualityReport:
    n_records: int
    field_accuracy_mean: float
    entity_recall_mean: float
    entity_precision_mean: float
    preference_f1_mean: float
    missing_field_quality_mean: float
    per_record: list[dict[str, Any]] = field(default_factory=list)


def _flatten_paths(d: dict, prefix: str = "") -> set[str]:
    """All leaf field paths of a nested dict (for accuracy denominator)."""
    paths: set[str] = set()
    for k, v in d.items():
        p = f"{prefix}.{k}" if prefix else k
        if isinstance(v, dict):
            paths |= _flatten_paths(v, p)
        elif isinstance(v, list):
            for i, item in enumerate(v):
                if isinstance(item, dict):
                    paths |= _flatten_paths(item, f"{p}[{i}]")
                else:
                    paths.add(f"{p}[{i}]")
        else:
            paths.add(p)
    return paths


def _compute_field_accuracy(draft: dict, edits: list[dict]) -> float:
    total = len(_flatten_paths(draft))
    if total == 0:
        return 1.0
    edited_paths = {e["field_path"] for e in edits}
    return 1.0 - (len(edited_paths) / total)


def _entity_set(packet: dict) -> set[tuple[str, str]]:
    return {
        (e["name"], e.get("ts_code") or "")
        for e in packet.get("chat_derived_signals", {}).get("entities", [])
    }


def _compute_entity_metrics(draft: dict, confirmed: dict) -> tuple[float, float]:
    """recall = |draft ∩ confirmed| / |confirmed|, precision = |draft ∩ confirmed| / |draft|."""
    d, c = _entity_set(draft), _entity_set(confirmed)
    if not c:
        return 1.0, 1.0
    inter = d & c
    recall = len(inter) / len(c)
    precision = len(inter) / len(d) if d else 0.0
    return recall, precision


def _preference_set(packet: dict) -> set[str]:
    return {
        p["text"]
        for p in packet.get("chat_derived_signals", {}).get("preferences", [])
    }


def _compute_preference_f1(draft: dict, confirmed: dict) -> float:
    d, c = _preference_set(draft), _preference_set(confirmed)
    if not c and not d:
        return 1.0
    inter = d & c
    p = len(inter) / len(d) if d else 0.0
    r = len(inter) / len(c) if c else 0.0
    if p + r == 0:
        return 0.0
    return 2 * p * r / (p + r)


def _compute_missing_field_quality(packet_draft: dict, packet_confirmed: dict) -> float:
    """1 if user filled in field that LLM marked missing; else 0."""
    hints = packet_draft.get("missing_field_hints", [])
    if not hints:
        return 1.0  # no hints, vacuously perfect
    n_correct = 0
    for h in hints:
        path = h["field_path"]
        # Check whether confirmed packet has non-empty value at that path
        value = _get_path(packet_confirmed, path)
        if value not in (None, "", [], {}):
            n_correct += 1
    return n_correct / len(hints)


def _get_path(d: dict, path: str) -> Any:
    cur: Any = d
    for part in path.split("."):
        if isinstance(cur, dict):
            cur = cur.get(part)
        else:
            return None
    return cur


async def run_extraction_quality_eval(database_url: str | None = None) -> ExtractionQualityReport:
    """Pull all EscalationRecord rows, compute metrics, return aggregated report."""
    dsn = database_url or os.environ["DATABASE_URL"]
    conn = await asyncpg.connect(dsn)
    rows = await conn.fetch(
        "SELECT id, packet_draft, packet_confirmed, user_edits FROM escalation_records "
        "WHERE status = 'completed'"
    )
    await conn.close()

    accs, e_recalls, e_precisions, p_f1s, m_qualities = [], [], [], [], []
    per_record = []

    for row in rows:
        draft = json.loads(row["packet_draft"])
        confirmed = json.loads(row["packet_confirmed"])
        edits = json.loads(row["user_edits"]) if row["user_edits"] else []

        acc = _compute_field_accuracy(draft, edits)
        er, ep = _compute_entity_metrics(draft, confirmed)
        pf1 = _compute_preference_f1(draft, confirmed)
        mq = _compute_missing_field_quality(draft, confirmed)

        accs.append(acc); e_recalls.append(er); e_precisions.append(ep)
        p_f1s.append(pf1); m_qualities.append(mq)
        per_record.append({
            "record_id": row["id"],
            "field_accuracy": acc,
            "entity_recall": er,
            "entity_precision": ep,
            "preference_f1": pf1,
            "missing_field_quality": mq,
        })

    def _mean(xs: list[float]) -> float:
        return sum(xs) / len(xs) if xs else 0.0

    return ExtractionQualityReport(
        n_records=len(rows),
        field_accuracy_mean=_mean(accs),
        entity_recall_mean=_mean(e_recalls),
        entity_precision_mean=_mean(e_precisions),
        preference_f1_mean=_mean(p_f1s),
        missing_field_quality_mean=_mean(m_qualities),
        per_record=per_record,
    )


if __name__ == "__main__":
    rep = asyncio.run(run_extraction_quality_eval())
    print(json.dumps({
        "n_records": rep.n_records,
        "field_accuracy_mean": rep.field_accuracy_mean,
        "entity_recall_mean": rep.entity_recall_mean,
        "entity_precision_mean": rep.entity_precision_mean,
        "preference_f1_mean": rep.preference_f1_mean,
        "missing_field_quality_mean": rep.missing_field_quality_mean,
    }, indent=2))
```
