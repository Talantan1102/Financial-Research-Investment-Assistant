```python
# backend/app/agents/plan_validator.py
"""Plan validator — three-layer gate for LLM-generated ResearchPlan (v1.x A4).

L1 schema (Pydantic — strict in LLMService).
L2 business rules (required dims / tool palette / total / dup).
L3 cost (optional, env-gated).

spec ref: docs/superpowers/specs/2026-05-15-v1.x-plan-template-validator-design.md § 4.3
"""

from __future__ import annotations

from pydantic import BaseModel, ConfigDict, Field

from app.agents.plan_template import DimensionSpec, PlanTemplate
from app.agents.schemas import ResearchPlan, Subtask


class ValidationResult(BaseModel):
    model_config = ConfigDict(frozen=True)

    ok: bool
    errors: list[str] = Field(default_factory=list)


def _dim_keyword_match(subtask: Subtask, dim_name: str) -> bool:
    """A subtask covers a dim if its description或rationale contains dim name."""
    return dim_name in subtask.description or dim_name in subtask.rationale


def _has_subtask_for_dim(plan: ResearchPlan, dim: DimensionSpec) -> bool:
    return any(_dim_keyword_match(s, dim["name"]) for s in plan.subtasks)


def _tool_count_for_dim(plan: ResearchPlan, dim: DimensionSpec) -> int:
    """Count unique tools in subtasks tagged with this dim, intersected with candidates."""
    candidates = set(dim["tool_candidates"])
    used: set[str] = set()
    for sub in plan.subtasks:
        if not _dim_keyword_match(sub, dim["name"]):
            continue
        for t in sub.required_tools:
            if t in candidates:
                used.add(t)
    return len(used)


def _all_tool_calls(plan: ResearchPlan) -> list[str]:
    return [t for sub in plan.subtasks for t in sub.required_tools]


def _has_duplicate_calls(plan: ResearchPlan) -> bool:
    """Same tool name appearing in 2+ subtasks counts as duplicate when forbid_duplicate=True."""
    seen: set[str] = set()
    for sub in plan.subtasks:
        for t in sub.required_tools:
            if t in seen:
                return True
            seen.add(t)
    return False


def validate_plan(plan: ResearchPlan, template: PlanTemplate) -> ValidationResult:
    errors: list[str] = []

    # L2.a — required dims coverage
    for dim in template["required_dimensions"]:
        if not _has_subtask_for_dim(plan, dim):
            errors.append(f"missing required dim: {dim['name']}")
            continue
        tool_count = _tool_count_for_dim(plan, dim)
        if tool_count < dim["min_tools"]:
            errors.append(
                f"dim '{dim['name']}' has < {dim['min_tools']} tools "
                f"(got {tool_count} from {dim['tool_candidates']})"
            )

    # L2.b — tool palette whitelist
    palette = set(template["tool_palette"])
    for sub in plan.subtasks:
        for t in sub.required_tools:
            if t not in palette:
                errors.append(f"tool '{t}' not in palette (LLM hallucinated)")

    # L2.c — subtask count
    if len(plan.subtasks) > template["constraints"]["max_subtasks"]:
        errors.append(
            f"too many subtasks: {len(plan.subtasks)} > "
            f"{template['constraints']['max_subtasks']}"
        )

    # L2.d — per-subtask tool count
    max_per = template["constraints"]["max_tool_calls_per_subtask"]
    for sub in plan.subtasks:
        if len(sub.required_tools) > max_per:
            errors.append(
                f"subtask '{sub.subtask_id}' has {len(sub.required_tools)} tools > {max_per}"
            )

    # L2.e — duplicate calls
    if template["constraints"]["forbid_duplicate_calls"] and _has_duplicate_calls(plan):
        errors.append("duplicate tool calls across subtasks detected")

    # L3 cost — wired in Task 1.3+ if needed; skip in v1.x ship
    return ValidationResult(ok=not errors, errors=errors)
```

```python
# backend/app/agents/research_planner.py
"""ResearchPlanner — v1.x template-guided generation + validator loop.

LLM emits a full ResearchPlan (description + rationale + tool selection per
subtask) within DD_PLAN_TEMPLATE constraints. Validator enforces; retry ≤2
then fallback to SAFE_DEFAULT_PLAN.

spec ref: docs/superpowers/specs/2026-05-15-v1.x-plan-template-validator-design.md § 4
"""

from __future__ import annotations

import json
from typing import Any

from app.agents.base import Agent
from app.agents.plan_template import DD_PLAN_TEMPLATE, SAFE_DEFAULT_PLAN, PlanTemplate
from app.agents.plan_validator import validate_plan
from app.agents.schemas import ResearchPlan, ResearchState, StepResult, Subtask
from app.services.llm_response import Tier

MAX_VALIDATOR_RETRY = 2

_PLANNER_SYSTEM_PROMPT_TEMPLATE = """你是金融研究助手 research_planner。\
你的工作: 基于客户画像 + 用户消息 + 下方 plan template, 生成一份 ResearchPlan。

# 客户画像 (6 字段)
- investment_objective: {objective}
- investment_horizon: {horizon}
- risk_tolerance: {risk}
- client_total_aum (CNY): {aum}
- client_existing_position (CNY): {position}
- target_ts_code: {ts_code}

# 用户消息
{user_message}

# Plan Template — 你必须遵守
## 必填维度 (required dimensions) — 每个必须产 ≥1 个 subtask
{required_dims_block}

## 可选维度 (optional dimensions) — 根据客户画像/消息决定是否产
{optional_dims_block}

## Tool 白名单 (palette)
{palette_block}

## 业务约束
- max_subtasks: {max_subtasks}
- max_tool_calls_per_subtask: {max_per_sub}
- forbid_duplicate_calls: 不要在多个 subtask 重复同一 tool

# Validator Feedback (上一轮失败时反馈)
{validator_feedback}

# 输出要求
仅输出 JSON,符合 ResearchPlan schema:
{{
  "rationale": "<≤300 字>",
  "subtasks": [
    {{"subtask_id": "s1", "description": "...", "required_tools": ["..."], "rationale": "..."}}
  ]
}}
- subtask_id 用 s1/s2/s3... 顺序
- description 标明它覆盖哪个维度(如开头写 "财务全景 — ...")
- required_tools 只能从 tool palette 选
- 维度覆盖检查会按 description/rationale 内是否包含维度名进行匹配
"""


def _format_required_dims(template: PlanTemplate) -> str:
    lines = []
    for d in template["required_dimensions"]:
        candidates = ", ".join(d["tool_candidates"])
        lines.append(f"- {d['name']}: candidates=[{candidates}], min_tools={d['min_tools']}")
    return "\n".join(lines)


def _format_optional_dims(template: PlanTemplate) -> str:
    lines = []
    for d in template["optional_dimensions"]:
        candidates = ", ".join(d["tool_candidates"])
        lines.append(f"- {d['name']}: candidates=[{candidates}]")
    return "\n".join(lines)


def build_planner_prompt(
    state: ResearchState,
    template: PlanTemplate,
    validator_feedback: str | None = None,
) -> str:
    user_msg = state.user_message.strip() if state.user_message else "(无)"
    feedback = (
        f"上一轮生成的 plan 未通过 Validator,errors:\n{validator_feedback}\n"
        f"请修正后重试。"
        if validator_feedback
        else "(无 — 首次生成)"
    )
    return _PLANNER_SYSTEM_PROMPT_TEMPLATE.format(
        objective=state.investment_objective or "(未指定)",
        horizon=state.investment_horizon or "(未指定)",
        risk=state.risk_tolerance or "(未指定)",
        aum=state.client_total_aum if state.client_total_aum is not None else "(未指定)",
        position=(
            state.client_existing_position
            if state.client_existing_position is not None
            else "(未指定)"
        ),
        ts_code=state.target_ts_code or "(未指定)",
        user_message=user_msg,
        required_dims_block=_format_required_dims(template),
        optional_dims_block=_format_optional_dims(template),
        palette_block=", ".join(template["tool_palette"]),
        max_subtasks=template["constraints"]["max_subtasks"],
        max_per_sub=template["constraints"]["max_tool_calls_per_subtask"],
        validator_feedback=feedback,
    )


def _build_safe_default_plan(target_name: str, ts_code: str) -> ResearchPlan:
    subtasks = [
        Subtask(
            subtask_id=f"safe_{idx}",
            description=t.description_template.format(target_name=target_name, ts_code=ts_code),
            required_tools=list(t.required_tools),
            rationale=t.rationale,
        )
        for idx, t in enumerate(SAFE_DEFAULT_PLAN)
    ]
    return ResearchPlan(
        rationale="Validator retry exhausted → SAFE_DEFAULT_PLAN fallback",
        subtasks=subtasks,
    )


class ResearchPlanner(Agent):
    name = "ResearchPlanner"
    model_tier: Tier = "balanced"
    template: PlanTemplate = DD_PLAN_TEMPLATE

    def step(self, state: ResearchState) -> StepResult:  # type: ignore[override]
        validator_feedback: str | None = None
        last_plan: ResearchPlan | None = None
        for attempt in range(1 + MAX_VALIDATOR_RETRY):
            prompt = build_planner_prompt(state, self.template, validator_feedback)
            r = self._llm.chat(
                prompt=prompt,
                tier=self.model_tier,
                schema=ResearchPlan,
                request_id=state.request_id,
            )
            assert isinstance(r.parsed, ResearchPlan)
            plan: ResearchPlan = r.parsed
            v = validate_plan(plan, self.template)
            if v.ok:
                return StepResult(
                    state_update={"plan": plan},
                    span_metadata={
                        "agent": "ResearchPlanner",
                        "model": r.model,
                        "cost_cny": r.cost_cny,
                        "validator_attempts": attempt + 1,
                        "validator_fallback": False,
                    },
                )
            last_plan = plan
            validator_feedback = "; ".join(v.errors)

        # Retry exhausted — fallback
        target_name = state.target_entity or state.target_ts_code or ""
        ts_code = state.target_ts_code or ""
        safe_plan = _build_safe_default_plan(target_name, ts_code)
        return StepResult(
            state_update={"plan": safe_plan},
            span_metadata={
                "agent": "ResearchPlanner",
                "validator_attempts": 1 + MAX_VALIDATOR_RETRY,
                "validator_fallback": True,
                "validator_last_errors": validator_feedback,
                "validator_last_plan": last_plan.model_dump() if last_plan else None,
            },
        )
```

```python
def _writer_retry_router(state: ResearchState) -> Literal["retry", "continue"]:
```

```python
# backend/app/agents/escalation_confidence.py
"""C6 Composite confidence — LLM self-report ∩ deterministic features, take min.

spec ref: docs/superpowers/specs/2026-05-15-v1.x-plan-template-validator-design.md § 6.5
"""

from __future__ import annotations

from typing import Any, Protocol

from app.agents.escalation_protocol import EscalationPacket

_LLM_CONF_MAP: dict[str, float] = {"high": 0.9, "medium": 0.6, "low": 0.3}

# Tuned defaults — adjustable via dogfood calibration (spec § 6.5 Step 4)
F_THRESHOLDS = {
    "chat_length_target": 5,        # F1: 5+ turns 给满分
    "entity_density_target": 3,     # F2: 3+ 次提及给满分
    "explicit_keywords": frozenset({
        "尽调", "深度分析", "详细看看", "全面研究", "深入研究", "做个调研",
    }),
}

# Lightweight chat message protocol — accommodate both ChatMessage Pydantic
# instances and test stubs without importing the full schema.
class _ChatMsgLike(Protocol):
    content: str


def _f1_chat_length(history: list[_ChatMsgLike]) -> float:
    return min(len(history) / F_THRESHOLDS["chat_length_target"], 1.0)


def _f2_entity_density(
    history: list[_ChatMsgLike], ts_code: str, target_name: str
) -> float:
    mentions = sum(
        1 for m in history if ts_code in m.content or (target_name and target_name in m.content)
    )
    return min(mentions / F_THRESHOLDS["entity_density_target"], 1.0)


def _f3_explicit_intent(history: list[_ChatMsgLike]) -> float:
    recent = history[-3:] if len(history) >= 3 else history
    kw = F_THRESHOLDS["explicit_keywords"]
    return 1.0 if any(k in m.content for m in recent for k in kw) else 0.6


def _has_contradiction(focus: list[str]) -> bool:
    """Naive contradiction heuristic — could be replaced with embedding check."""
    PAIRS = [
        ("买入", "卖出"), ("加仓", "减仓"),
        ("做多", "做空"), ("看多", "看空"),
        ("乐观", "悲观"),
    ]
    joined = " ".join(focus)
    return any(a in joined and b in joined for a, b in PAIRS)


def _f4_no_contradiction(extracted: EscalationPacket) -> float:
    return 0.3 if _has_contradiction(extracted.discussion_focus) else 1.0


def _f5_explicit_confirm(user_confirmed: bool) -> float:
    return 1.0 if user_confirmed else 0.7


def compute_deterministic_confidence(
    extracted: EscalationPacket,
    chat_history: list[Any],  # list[_ChatMsgLike] — Any for caller flexibility
    target_ts_code: str,
    target_name: str,
    user_confirmed_escalation: bool,
) -> float:
    features = {
        "chat_length": _f1_chat_length(chat_history),
        "entity_density": _f2_entity_density(chat_history, target_ts_code, target_name),
        "explicit_intent": _f3_explicit_intent(chat_history),
        "no_contradiction": _f4_no_contradiction(extracted),
        "explicit_confirm": _f5_explicit_confirm(user_confirmed_escalation),
    }
    return sum(features.values()) / len(features)


def compute_confidence(
    extracted: EscalationPacket,
    chat_history: list[Any],
    target_ts_code: str,
    target_name: str,
    user_confirmed_escalation: bool,
) -> float:
    """Composite — min(LLM self, deterministic). Fail-safe金融场景保守取低。"""
    llm = _LLM_CONF_MAP[extracted.llm_self_confidence]
    det = compute_deterministic_confidence(
        extracted, chat_history, target_ts_code, target_name, user_confirmed_escalation,
    )
    return min(llm, det)
```

```python
def _format_distilled_block(state: ResearchState) -> str:
    """Multi-turn distilled signals — empty string when no distillation present."""
    if not state.escalation_intent and not state.discussion_focus and not state.explicit_exclusions:
```

```python
async def _build_research_state_from_escalation(
    packet: EscalationPacket,
    chat_history: list[ChatMessage],
    settings: Settings,
    llm_service: LLMService,
) -> ResearchState:
    """v1.x fallback ladder:
    1. confidence >= threshold → use distilled fields
    2. confidence < threshold → summarize last 3 turns to user_message
    3. summary fails → form-style fallback (user_message = raw last turn)
    """
    from app.agents.escalation_confidence import compute_confidence
    
    conf = compute_confidence(
        extracted=packet,
        chat_history=chat_history,
        target_ts_code=packet.explicit_task.target_ts_code or "",
        target_name=packet.explicit_task.target_entity_name or "",
        user_confirmed_escalation=True,  # entry from confirm button
    )

    if conf >= settings.ESCALATION_CONFIDENCE_THRESHOLD:
        return ResearchState(
            user_id=...,
            session_id=...,
            user_message=packet.explicit_task.user_extra_message or "",
            request_id=...,
            escalation_intent=packet.escalation_intent,
            discussion_focus=packet.discussion_focus,
            explicit_exclusions=packet.explicit_exclusions,
            # ... other fields from packet ...
        )

    # Layer 1 fallback — summarize last 3 turns
    try:
        summary = await _summarize_recent_chat(chat_history[-3:], llm_service)
        return ResearchState(
            user_id=..., session_id=..., request_id=...,
            user_message=summary,
            # distilled fields left empty — falls to form-style downstream
        )
    except Exception:  # noqa: BLE001  log + fallback further
        # Layer 2 fallback — form-style (raw last turn)
        return ResearchState(
            user_id=..., session_id=..., request_id=...,
            user_message=packet.explicit_task.raw_last_user_turn,

        + "\n".join(f"[{m.role}] {m.content}" for m in msgs)
    )
```
