# v0.9.x 前端重构 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 前端从 v0~v0.5 残留 + 投资尽调缺展示页的混乱状态,重构为研报展示 + 历史列表 + auth 改写的对齐形态;后端补 `research_reports` 持久化层 + 5 个新 endpoint;legacy PG 硬依赖一并解决。

**Architecture:** 前端 React 19 + antd 5 + valtio + react-router-dom 6;新建 `/research` 列表 + `/research/:id` 展示页(D 形态:PDF-like 长卷 + 流式 sticky progress overlay);删 4 路由 + 改写 auth + 新建 register。后端 SQLAlchemy → legacy PG 加 `research_reports` 表 + 5 endpoint(`/api/reports` CRUD + `/api/reports/:id/stream`);auth 与 chat 解耦。docker-compose PG + CI 跑通作为前置。

**Tech Stack:** React 19 + TypeScript 5.7 + Vite + antd 5 + valtio + react-router-dom 6 + axios + marked + echarts;FastAPI + SQLAlchemy 2 + PostgreSQL 14+ + LangGraph 1.x(已有);docker-compose;pytest + ruff + mypy strict / vitest 或 jest 走前端 lint 链。

**Spec ref:** `docs/superpowers/specs/2026-05-06-v0.9.x-frontend-rebuild-design.md`

```python
from fastapi import APIRouter, HTTPException, status
from pydantic import BaseModel, EmailStr
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

# 依赖:passlib bcrypt 假设 legacy 已用;如果没有,加:
# uv add passlib[bcrypt]
from passlib.context import CryptContext

from app.core.database import get_db
from app.models.user import User
# JWT 工具假设 legacy 已有;否则查现状

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


class RegisterRequest(BaseModel):
    username: str
    password: str
    email: EmailStr


class RegisterResponse(BaseModel):
    token: str
    user: dict


@router.post("/register", response_model=RegisterResponse, status_code=status.HTTP_201_CREATED)
async def register(payload: RegisterRequest, db: Session = Depends(get_db)) -> RegisterResponse:
    """注册新用户.
    
    Validation:
    - username/email unique
    - password bcrypt 哈希存储
    - 注册后自动签发 token(等同 login)
    """
    hashed = pwd_context.hash(payload.password)
    user = User(
        username=payload.username,
        email=payload.email,
        hashed_password=hashed,
        is_active=True,
    )
    try:
        db.add(user)
        db.commit()
        db.refresh(user)
    except IntegrityError:
        db.rollback()
        raise HTTPException(status_code=400, detail="Username 或 email 已存在")
    
    token = create_access_token(user_id=str(user.id))  # 调 legacy JWT 工具
    return RegisterResponse(
        token=token,
        user={
            "id": str(user.id),
            "username": user.username,
            "email": user.email,
            "is_active": user.is_active,
            "created_at": user.created_at.isoformat(),
        },
    )
```

```python
# backend/app/router/reports.py
"""POST/GET/DELETE /api/reports — 研报 CRUD + SSE."""
from __future__ import annotations

from datetime import datetime
from typing import Annotated, Any
from uuid import uuid4

from fastapi import APIRouter, Depends, HTTPException, Query, status
from pydantic import BaseModel
from sqlalchemy.orm import Session

from app.core.database import get_db
from app.models.research_report import ResearchReport
from app.router.auth_helpers import _AnonUser, get_current_user

router = APIRouter(prefix="/api/reports", tags=["reports-v0.9"])


class ReportListItem(BaseModel):
    id: str
    target_name: str
    target_ts_code: str | None
    status: str
    cost: float
    created_at: datetime
    investment_recommendation: str | None  # 从 report_json 提取


class ReportListResponse(BaseModel):
    items: list[ReportListItem]
    total: int
    page: int
    page_size: int


class ReportDetailResponse(BaseModel):
    id: str
    target_name: str
    target_ts_code: str | None
    status: str
    cost: float
    created_at: datetime
    updated_at: datetime
    request_id: str | None
    report_json: dict[str, Any]


class ReportStartRequest(BaseModel):
    target_name: str
    target_ts_code: str | None = None
    research_style: str = "comprehensive"


class ReportStartResponse(BaseModel):
    id: str  # 前端拿这个 id 后跳转 /research/:id 订阅 SSE


@router.get("", response_model=ReportListResponse)
async def list_reports(
    user: Annotated[_AnonUser, Depends(get_current_user)],
    db: Session = Depends(get_db),
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=100),
) -> ReportListResponse:
    """当前 user 的历史研报,paginated."""
    q = db.query(ResearchReport).filter(ResearchReport.user_id == user.id)
    total = q.count()
    rows = q.order_by(ResearchReport.created_at.desc()).offset((page - 1) * page_size).limit(page_size).all()
    items = [
        ReportListItem(
            id=r.id,
            target_name=r.target_name,
            target_ts_code=r.target_ts_code,
            status=r.status,
            cost=float(r.cost),
            created_at=r.created_at,
            investment_recommendation=(
                (r.report_json or {}).get("investment_recommendation", {}).get("recommendation")
            ),
        )
        for r in rows
    ]
    return ReportListResponse(items=items, total=total, page=page, page_size=page_size)


@router.get("/{report_id}", response_model=ReportDetailResponse)
async def get_report(
    report_id: str,
    user: Annotated[_AnonUser, Depends(get_current_user)],
    db: Session = Depends(get_db),
) -> ReportDetailResponse:
    r = db.query(ResearchReport).filter(ResearchReport.id == report_id).first()
    if r is None:
        raise HTTPException(status_code=404, detail="Report not found")
    if r.user_id != user.id:
        raise HTTPException(status_code=403, detail="Not your report")
    return ReportDetailResponse(
        id=r.id,
        target_name=r.target_name,
        target_ts_code=r.target_ts_code,
        status=r.status,
        cost=float(r.cost),
        created_at=r.created_at,
        updated_at=r.updated_at,
        request_id=r.request_id,
        report_json=r.report_json,
    )


@router.delete("/{report_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_report(
    report_id: str,
    user: Annotated[_AnonUser, Depends(get_current_user)],
    db: Session = Depends(get_db),
) -> None:
    r = db.query(ResearchReport).filter(ResearchReport.id == report_id).first()
    if r is None:
        raise HTTPException(status_code=404, detail="Report not found")
    if r.user_id != user.id:
        raise HTTPException(status_code=403, detail="Not your report")
    db.delete(r)
    db.commit()


@router.post("", response_model=ReportStartResponse, status_code=status.HTTP_201_CREATED)
async def start_report(
    payload: ReportStartRequest,
    user: Annotated[_AnonUser, Depends(get_current_user)],
    db: Session = Depends(get_db),
) -> ReportStartResponse:
    """启动新研报 — 创建 placeholder row(status=streaming),返回 id 给前端订阅 SSE."""
    report_id = str(uuid4())
    r = ResearchReport(
        id=report_id,
        user_id=user.id,
        target_name=payload.target_name,
        target_ts_code=payload.target_ts_code,
        status="streaming",
        report_json={},  # 流式生成中,完成后更新
        cost=0,
    )
    db.add(r)
    db.commit()
    return ReportStartResponse(id=report_id)
```

```python
@router.get("/api/reports/{report_id}/stream")
async def stream_report(
    report_id: str,
    user: Annotated[_AnonUser, Depends(get_current_user)],
    db: Session = Depends(get_db),
    graph: Any = Depends(get_research_graph),
) -> StreamingResponse:
    """订阅指定研报的 SSE 流."""
    r = db.query(ResearchReport).filter(ResearchReport.id == report_id).first()
    if r is None:
        raise HTTPException(status_code=404, detail="Report not found")
    if r.user_id != user.id:
        raise HTTPException(status_code=403, detail="Not your report")
    
    req = ResearchRequest(
        target_entity=r.target_name,
        user_message=f"研究 {r.target_name}",
        research_style="comprehensive",
    )
    return StreamingResponse(
        _stream_research_with_persist(req, user, graph, r, db),
        media_type="text/event-stream",
    )


async def _stream_research_with_persist(
    req: ResearchRequest, user: _AnonUser, graph: Any, report: ResearchReport, db: Session
) -> AsyncIterator[str]:
    """复用 _stream_research,但 done event 时把 final report_json 写回 db."""
    final_report: dict[str, Any] = {}
    total_cost = 0.0
    try:
        async for sse_line in _stream_research(req, user, graph):
            yield sse_line
            # 解析 event,如果 type=report_chunk 累积 final_report;type=done 写入 db
            try:
                event_data = json.loads(sse_line.removeprefix("data: ").strip())
                if event_data.get("type") == "report_chunk":
                    final_report.update(event_data.get("data", {}))
                elif event_data.get("type") == "critic_score":
                    total_cost += event_data.get("data", {}).get("cost", 0.0)
                elif event_data.get("type") == "done":
                    report.report_json = final_report
                    report.status = "completed"
                    report.cost = total_cost
                    db.commit()
            except Exception:
                pass
    except Exception as e:
        report.status = "failed"
        report.report_json = {"error": str(e)}
        db.commit()
        raise
```

```ts
export type TimeFilter = 'today' | 'week' | 'month' | 'all'

export interface ProgressEvent {
  type: 'plan' | 'data_progress' | 'insight' | 'report_chunk' | 'critic_score' | 'done' | 'error'
  message: string
  timestamp: number
}

interface ReportState {
  // 列表态
  list: ReportListItem[]
  listTotal: number
  listLoading: boolean
  timeFilter: TimeFilter

  // 详情态
  current: ReportDetail | null
  currentLoading: boolean

  // 流式态
  streaming: {
    active: boolean
    progress: ProgressEvent[]
    partialSections: Record<string, any>  // section_name → data
  }
}

export const reportState = proxy<ReportState>({
  list: [],
  listTotal: 0,
  listLoading: false,
  timeFilter: 'all',
  current: null,
  currentLoading: false,
  streaming: {
    active: false,
    progress: [],
    partialSections: {},
  },
})

export const reportActions = {
  async fetchList(page = 1) {
    reportState.listLoading = true
    try {
      const res = await reportsApi.listReports(page)
      reportState.list = res.items
      reportState.listTotal = res.total
    } finally {
      reportState.listLoading = false
    }
  },

  async fetchDetail(id: string) {
    reportState.currentLoading = true
    try {
      const res = await reportsApi.getReport(id)
      reportState.current = res
    } finally {
      reportState.currentLoading = false
    }
  },

  async deleteReport(id: string) {
    await reportsApi.deleteReport(id)
    reportState.list = reportState.list.filter(r => r.id !== id)
    reportState.listTotal -= 1
  },

  async startReport(targetName: string, tsCode?: string) {
    const res = await reportsApi.startReport({
      target_name: targetName,
      target_ts_code: tsCode,
    })
    return res.id
  },

  startStreaming(reportId: string, token: string) {
    reportState.streaming.active = true
    reportState.streaming.progress = []
    reportState.streaming.partialSections = {}

    const url = `/api/reports/${reportId}/stream`
    const eventSource = new EventSource(url, { withCredentials: true })
    // 注:EventSource 不直接支持 Authorization header,实施时考虑用 query string 或 fetch + ReadableStream
    // 这里 sketch 形态,实施按真实 backend SSE auth 调用方式实现

    eventSource.onmessage = (e) => {
      try {
        const event = JSON.parse(e.data)
        reportState.streaming.progress.push({
          type: event.type,
          message: event.data?.message ?? '',
          timestamp: Date.now(),
        })
        if (event.type === 'report_chunk') {
          Object.assign(reportState.streaming.partialSections, event.data)
        }
        if (event.type === 'done') {
          reportState.streaming.active = false
          eventSource.close()
          // 拉取最终 report
          reportActions.fetchDetail(reportId)
        }
        if (event.type === 'error') {
          reportState.streaming.active = false
          eventSource.close()
        }
      } catch (err) {
        console.error('parse SSE event error', err)
      }
    }

    eventSource.onerror = () => {
      reportState.streaming.active = false
      eventSource.close()
    }

    return () => {
      eventSource.close()
      reportState.streaming.active = false
    }
  },

  setTimeFilter(filter: TimeFilter) {
    reportState.timeFilter = filter
  },
}
```

```tsx
// frontend/src/components/report-canvas/index.tsx
import { Anchor, Typography } from 'antd'
import { marked } from 'marked'
import type { InvestmentDueDiligenceReport } from '@/types/research'
import styles from './index.module.scss'

interface Props {
  report: InvestmentDueDiligenceReport
}

interface SectionDef {
  key: string
  title: string
  data: any
}

const SECTION_ORDER: { key: keyof InvestmentDueDiligenceReport; title: string }[] = [
  { key: 'target_overview', title: '一、标的概览' },
  { key: 'legal_qualification', title: '二、合规与资质' },
  { key: 'financial_health', title: '三、财务健康' },
  { key: 'industry_position', title: '四、行业地位' },
  { key: 'risk_factors', title: '五、风险因素' },
  { key: 'investment_recommendation', title: '六、投资建议' },
]

function renderSection(data: any): string {
  /** 把 section data 转 markdown(简化:JSON.stringify 转表格,实施时按字段语义渲染) */
  if (!data) return '_(此节无数据)_'
  if (typeof data === 'string') return data
  return Object.entries(data)
    .map(([k, v]) => `- **${k}**: ${typeof v === 'object' ? JSON.stringify(v) : v}`)
    .join('\n')
}

export default function ReportCanvas({ report }: Props) {
  const sections = SECTION_ORDER.filter((s) => report[s.key]).map((s) => ({
    ...s,
    data: report[s.key],
  }))

  return (
    <div className={styles.container}>
      <div className={styles.canvas}>
        {sections.map((s) => (
          <section key={s.key} id={s.key} className={styles.section}>
            <Typography.Title level={2}>{s.title}</Typography.Title>
            <div
              className={styles.markdown}
              dangerouslySetInnerHTML={{ __html: marked.parse(renderSection(s.data)) as string }}
            />
          </section>
        ))}
      </div>
      <aside className={styles.toc}>
        <Anchor
          items={sections.map((s) => ({ key: s.key, href: `#${s.key}`, title: s.title }))}
          targetOffset={80}
        />
      </aside>
    </div>
  )
}
```

```ts
// frontend/src/utils/report-to-markdown.ts
import type { InvestmentDueDiligenceReport } from '@/types/research'

const SECTION_TITLES: Record<string, string> = {
  target_overview: '一、标的概览',
  legal_qualification: '二、合规与资质',
  financial_health: '三、财务健康',
  industry_position: '四、行业地位',
  risk_factors: '五、风险因素',
  investment_recommendation: '六、投资建议',
}

function sectionToMarkdown(data: any): string {
  if (!data) return '_(此节无数据)_'
  if (typeof data === 'string') return data
  return Object.entries(data)
    .map(([k, v]) => `- **${k}**: ${typeof v === 'object' ? JSON.stringify(v) : v}`)
    .join('\n')
}

export function reportToMarkdown(report: InvestmentDueDiligenceReport, meta: { target_name: string; created_at: string; cost: number }): string {
  const sections = Object.entries(SECTION_TITLES)
    .filter(([k]) => (report as any)[k])
    .map(([k, title]) => `## ${title}\n\n${sectionToMarkdown((report as any)[k])}`)
    .join('\n\n---\n\n')

  return `# ${meta.target_name} 投资尽调研报\n\n_生成于 ${meta.created_at}, 成本 ¥${meta.cost.toFixed(2)}_\n\n${sections}\n`
}
```

```tsx
// frontend/src/components/export-button/index.tsx
import { Button, message } from 'antd'
import { CopyOutlined, FilePdfOutlined } from '@ant-design/icons'
import { reportToMarkdown } from '@/utils/report-to-markdown'
import type { ReportDetail } from '@/api/reports'

interface Props {
  report: ReportDetail
}

export default function ExportButton({ report }: Props) {
  const handleCopyMarkdown = async () => {
    const md = reportToMarkdown(report.report_json, {
      target_name: report.target_name,
      created_at: report.created_at,
      cost: report.cost,
    })
    await navigator.clipboard.writeText(md)
    message.success('Markdown 已复制到剪贴板')
  }

  const handlePrintPDF = () => {
    message.info('请在浏览器对话框中选择"另存为 PDF"', 1.5)
    setTimeout(() => window.print(), 1500)
  }

  return (
    <span>
      <Button icon={<CopyOutlined />} onClick={handleCopyMarkdown}>复制 Markdown</Button>
      <Button icon={<FilePdfOutlined />} onClick={handlePrintPDF} style={{ marginLeft: 8 }}>导出 PDF</Button>
    </span>
  )
}
```

```tsx
import { useState } from 'react'
import { Input, Button, Card } from 'antd'
import { useNavigate } from 'react-router-dom'
import { reportActions } from '@/store/report'

function LandingResearchEntry() {
  const [target, setTarget] = useState('')
  const [tsCode, setTsCode] = useState('')
  const navigate = useNavigate()

  const handleStart = async () => {
    if (!target.trim()) return
    const id = await reportActions.startReport(target.trim(), tsCode.trim() || undefined)
    navigate(`/research/${id}`)
  }

  return (
    <Card title="新建投资尽调研报" style={{ maxWidth: 600, margin: '32px auto' }}>
      <Input
        placeholder="目标名称(如 贵州茅台)"
        value={target}
        onChange={(e) => setTarget(e.target.value)}
        size="large"
        style={{ marginBottom: 12 }}
      />
      <Input
        placeholder="股票代码(如 600519.SH,可选)"
        value={tsCode}
        onChange={(e) => setTsCode(e.target.value)}
        size="large"
        style={{ marginBottom: 16 }}
      />
      <Button type="primary" size="large" block onClick={handleStart} disabled={!target.trim()}>
        开始研究
      </Button>
      <a onClick={() => navigate('/research')} style={{ display: 'block', textAlign: 'center', marginTop: 16 }}>
        查看历史研报 →
      </a>
    </Card>
  )
}
```
