**Goal:** Refactor v0.8.4 plan-driven Planner into a constrained LLM router (4 plan_id 选 1) backed by a hardcoded `plan_registry`, plus an Anthropic Skills bundle (`financial_research`, 17 components) with deterministic Python helpers for position size / recommendation classification, plus a Critic 第 7 scorer (`plan_correctness`) and LangGraph self-correcting retry edge (max 2 轮).

```python
async def get_cashflow(self, *, ts_code: str, end_date: str | None = None) -> pd.DataFrame:
    return await self._call_cached(
        "cashflow",
        {"ts_code": ts_code, "end_date": end_date or "20241231"},
    )

async def get_daily_basic(
    self, *, ts_code: str, trade_date: str | None = None
) -> pd.DataFrame:
    return await self._call_cached(
        "daily_basic",
        {"ts_code": ts_code, "trade_date": trade_date or self._latest_trade_date()},
    )

async def get_pe_history(
    self, *, ts_code: str, years_back: int = 5, current_pe: float | None = None
) -> pd.DataFrame:
    """聚合历史 daily_basic 计算 PE 分位.

    实现:取近 N 年的 daily_basic.pe 分布,计算 current_pe 在分布中的 percentile.
    """
    end = self._today_yyyymmdd()
    start = self._n_years_ago(years_back)
    history = await self._call_cached(
        "daily_basic",
        {"ts_code": ts_code, "start_date": start, "end_date": end, "fields": "pe"},
    )
    if current_pe is None:
        # 用最新一条作为 current
        latest = await self._call_cached(
            "daily_basic", {"ts_code": ts_code, "trade_date": end, "fields": "pe"}
        )
        current_pe = float(latest["pe"].iloc[0]) if not latest.empty else 0.0
    pe_series = history["pe"].dropna().sort_values()
    n = len(pe_series)
    rank = (pe_series < current_pe).sum()
    percentile = rank / max(n, 1)
    return pd.DataFrame(
        {
            "ts_code": [ts_code],
            "current_pe": [current_pe],
            "historical_percentile": [float(percentile)],
            "min_pe": [float(pe_series.min()) if n > 0 else 0.0],
            "max_pe": [float(pe_series.max()) if n > 0 else 0.0],
            "median_pe": [float(pe_series.median()) if n > 0 else 0.0],
        }
    )

async def get_forecast(self, *, ts_code: str, period: str | None = None) -> pd.DataFrame:
    return await self._call_cached(
        "forecast", {"ts_code": ts_code, "period": period or "20241231"}
    )

async def get_dividend_history(self, *, ts_code: str, years_back: int = 5) -> pd.DataFrame:
    end = self._today_yyyymmdd()
    start = self._n_years_ago(years_back)
    return await self._call_cached(
        "dividend",
        {"ts_code": ts_code, "ann_date_start": start, "ann_date_end": end},
    )

async def get_holder_change(self, *, ts_code: str, years_back: int = 2) -> pd.DataFrame:
    end = self._today_yyyymmdd()
    start = self._n_years_ago(years_back)
    return await self._call_cached(
        "stk_holdernumber",
        {"ts_code": ts_code, "start_date": start, "end_date": end},
    )

async def get_money_flow(
    self, *, ts_code: str, start_date: str, end_date: str
) -> pd.DataFrame:
    return await self._call_cached(
        "moneyflow",
        {"ts_code": ts_code, "start_date": start_date, "end_date": end_date},
    )
```

注:`_today_yyyymmdd()` 和 `_n_years_ago(n)` 是辅助 method。如果不存在,加到 RealTushareService 内部:

```python
def _today_yyyymmdd(self) -> str:
    from datetime import datetime, timezone
    return datetime.now(timezone.utc).strftime("%Y%m%d")

def _n_years_ago(self, n: int) -> str:
    from datetime import datetime, timedelta, timezone
    return (datetime.now(timezone.utc) - timedelta(days=365 * n)).strftime("%Y%m%d")
```

```python
# backend/app/tools/get_balance_sheet.py
"""GetBalanceSheetTool — 资产负债 + 派生偿债指标."""
from __future__ import annotations

from typing import Any

from pydantic import BaseModel

from app.services.tushare_service import TushareService, build_tushare_service
from app.tools.base import Tool


class BalanceSheetArgs(BaseModel):
    ts_code: str
    end_date: str | None = None


class GetBalanceSheetTool(Tool):
    name = "get_balance_sheet"
    description = (
        "Return total_assets, total_liab, total_cur_assets, total_cur_liab "
        "and derived asset_liability_ratio + current_ratio for an A-share."
    )
    args_schema = BalanceSheetArgs

    def __init__(self, tushare: TushareService | None = None) -> None:
        self._tushare = tushare or build_tushare_service()

    async def run(self, args: BaseModel) -> dict[str, Any]:
        a = args if isinstance(args, BalanceSheetArgs) else BalanceSheetArgs(**args.model_dump())
        df = await self._tushare.get_balance_sheet(ts_code=a.ts_code, end_date=a.end_date)
        if df.empty:
            return {"ts_code": a.ts_code, "error": "no data"}
        row = df.iloc[0]
        total_assets = float(row["total_assets"])
        total_liab = float(row["total_liab"])
        total_cur_assets = float(row["total_cur_assets"])
        total_cur_liab = float(row["total_cur_liab"])
        return {
            "ts_code": a.ts_code,
            "end_date": str(row["end_date"]),
            "total_assets": total_assets,
            "total_liab": total_liab,
            "total_cur_assets": total_cur_assets,
            "total_cur_liab": total_cur_liab,
            "asset_liability_ratio": total_liab / max(total_assets, 1.0),
            "current_ratio": total_cur_assets / max(total_cur_liab, 1.0),
        }
```

```python
# backend/app/skills/financial_research/scripts/compute_position_size.py
"""Deterministic position size calculator (v0.8.5 spec § 4.2)."""
from __future__ import annotations

from pathlib import Path
from typing import Literal

import yaml

_RULES_PATH = Path(__file__).parent.parent / "references" / "position_size_rules.yaml"
_RULES = yaml.safe_load(_RULES_PATH.read_text(encoding="utf-8"))


Recommendation = Literal[
    "recommend_buy", "recommend_overweight", "recommend_hold",
    "recommend_underweight", "recommend_sell",
]
RiskTolerance = Literal[
    "conservative", "moderate", "balanced", "aggressive", "very_aggressive",
]


def compute_position_size_pct(
    *,
    recommendation: Recommendation,
    risk_tolerance: RiskTolerance,
    market_cap_cny: float,
) -> float:
    """Pure function. Same input → same output. Deterministic."""
    base = float(_RULES["base_pct"][recommendation])
    multiplier = float(_RULES["risk_multiplier"][risk_tolerance])
    haircut = (
        float(_RULES["small_cap_haircut"])
        if market_cap_cny < float(_RULES["small_cap_threshold_cny"])
        else 1.0
    )
    pct = base * multiplier * haircut
    return min(pct, float(_RULES["max_position_pct"]))
```

```python
"""Industry benchmark lookup — JSON-backed deterministic query."""
from __future__ import annotations

import json
from pathlib import Path

_BENCHMARKS_PATH = Path(__file__).parent.parent / "references" / "industry_benchmarks.json"
_BENCHMARKS = json.loads(_BENCHMARKS_PATH.read_text(encoding="utf-8"))


def lookup_industry_benchmark(*, industry: str, indicator: str) -> float:
    """Return benchmark for given industry × indicator. Falls back to DEFAULT."""
    industry_data = _BENCHMARKS.get(industry, _BENCHMARKS["DEFAULT"])
    if indicator not in industry_data:
        # 该 industry 不含此 indicator → 用 DEFAULT
        return float(_BENCHMARKS["DEFAULT"][indicator])
    return float(industry_data[indicator])
```

```python
# backend/app/skills/financial_research/scripts/classify_recommendation.py
"""5-tier recommendation classifier — YAML-rules based."""
from __future__ import annotations

from pathlib import Path
from typing import Any, Literal

import yaml

_RULES_PATH = Path(__file__).parent.parent / "references" / "recommendation_rules.yaml"
_RULES = yaml.safe_load(_RULES_PATH.read_text(encoding="utf-8"))

Recommendation = Literal[
    "recommend_buy", "recommend_overweight", "recommend_hold",
    "recommend_underweight", "recommend_sell",
]

_OPS = {
    "<": lambda a, b: a < b,
    ">": lambda a, b: a > b,
    "==": lambda a, b: a == b,
    "<=": lambda a, b: a <= b,
    ">=": lambda a, b: a >= b,
}


def _eval_condition(cond: dict[str, Any], metrics: dict[str, Any]) -> bool:
    if "field" in cond:
        val = metrics.get(cond["field"])
        if val is None:
            return False
        return bool(_OPS[cond["op"]](val, cond["value"]))
    if "count_at_least_3" in cond:
        return sum(_eval_condition(c, metrics) for c in cond["count_at_least_3"]) >= 3
    return False


def _eval_rule(rule: dict[str, Any], metrics: dict[str, Any]) -> bool:
    conds = rule.get("conditions", {})
    if rule.get("fallback"):
        return True
    if "all_of" in conds:
        return all(_eval_condition(c, metrics) for c in conds["all_of"])
    if "any_of" in conds:
        return any(_eval_condition(c, metrics) for c in conds["any_of"])
    return False


# 评级优先级顺序(先 sell, 再 buy / overweight, 最后 underweight, 兜底 hold)
_PRIORITY: list[Recommendation] = [
    "recommend_sell",
    "recommend_buy",
    "recommend_overweight",
    "recommend_underweight",
    "recommend_hold",
]


def classify_recommendation(metrics: dict[str, Any]) -> Recommendation:
    """Pure function. metrics → recommendation enum."""
    rules = _RULES["rules"]
    for rec in _PRIORITY:
        if _eval_rule(rules[rec], metrics):
            return rec
    return "recommend_hold"  # 兜底
```

```python
# backend/app/skills/financial_research/__init__.py
"""financial_research skill bundle (v0.8.5).

Loads at module import time:
- 11 methodology markdown → dict[str, str]
- 3 references (json/yaml) → dict[str, dict | list]
- 3 Python helper modules → from .scripts namespace
"""
from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path
from types import ModuleType
from typing import Any

import yaml

from app.skills.financial_research import scripts as _scripts

_SKILL_DIR = Path(__file__).parent

_METHODOLOGY: dict[str, str] = {
    f.stem: f.read_text(encoding="utf-8")
    for f in (_SKILL_DIR / "methodology").glob("*.md")
}

_REFERENCES: dict[str, Any] = {
    "industry_benchmarks": json.loads(
        (_SKILL_DIR / "references" / "industry_benchmarks.json").read_text(encoding="utf-8")
    ),
    "recommendation_rules": yaml.safe_load(
        (_SKILL_DIR / "references" / "recommendation_rules.yaml").read_text(encoding="utf-8")
    ),
    "position_size_rules": yaml.safe_load(
        (_SKILL_DIR / "references" / "position_size_rules.yaml").read_text(encoding="utf-8")
    ),
}


@dataclass
class SkillBundle:
    methodology: dict[str, str]
    references: dict[str, Any]
    scripts: ModuleType

    def composed_sop(self) -> str:
        """Return concat of all 11 methodology markdown for prompt injection."""
        order = [
            "solvency", "profitability", "growth", "cashflow_quality",
            "valuation", "industry", "shareholder_governance",
            "short_term_capital_flow", "event_driven", "risk_factors",
            "decision_framework",
        ]
        return "\n\n".join(self.methodology[name] for name in order if name in self.methodology)


def load_skill() -> SkillBundle:
    """Module-level cached load (reads files only once at first import)."""
    return SkillBundle(
        methodology=_METHODOLOGY,
        references=_REFERENCES,
        scripts=_scripts,
    )
```

```python
# backend/app/agents/writer.py
from app.skills.financial_research.scripts import (
    classify_recommendation,
    compute_position_size_pct,
)


def post_process_writer_output(state: ResearchState, llm_report: dict) -> dict:
    """Override LLM-generated recommendation + position_size with deterministic Python helpers.

    LLM 仍 narrative; recommendation 5 档分类 + 仓位百分比由 Python 算。
    """
    # 提取 LLM 输出的 metrics (从 financial_analysis / valuation_analysis sections)
    metrics = _extract_metrics_from_llm_report(llm_report)

    # Override recommendation
    classified_rec = classify_recommendation(metrics)
    llm_report["investment_recommendation"]["recommendation"] = classified_rec

    # Override position size
    market_cap = metrics.get("market_cap_cny", 0.0)
    pct = compute_position_size_pct(
        recommendation=classified_rec,
        risk_tolerance=state.risk_tolerance,
        market_cap_cny=market_cap,
    )
    llm_report["investment_recommendation"]["recommended_position_size_pct"] = pct
    return llm_report


def _extract_metrics_from_llm_report(report: dict) -> dict:
    """Extract numerical metrics for classify_recommendation rule eval."""
    fa = report.get("financial_analysis") or {}
    va = fa.get("valuation_analysis") or {}
    overview = report.get("target_overview") or {}
    return {
        "pe_percentile": va.get("pe_historical_percentile") or 0.5,
        "roe": fa.get("roe") or 0.0,
        "revenue_yoy": fa.get("revenue_yoy") or 0.0,
        "net_profit_yoy": fa.get("net_profit_yoy") or 0.0,
        "market_cap_cny": overview.get("current_market_cap") or 0.0,
        "forecast_signal": fa.get("forecast_signal") or "neutral",
        "asset_liability_warning": fa.get("debt_ratio_assessment") in {"警戒", "高风险"},
    }
```

```python
_PLANNER_SYSTEM_PROMPT_TEMPLATE = """你是投资研究项目经理, 任务是根据客户输入选择最合适的研究 plan_id。

# 可选 plan_id (4 选 1):
- capital_preservation: 保本型客户 — 重点偿债能力 / 现金流 / 估值不高 / 风险因素
- stable_growth: 稳健增长 — 重点 ROE 稳健 / 分红连续性 / 行业龙头 / 偿债基础
- balanced: 平衡型 — 财务全面 / 估值合理 / 行业催化 / 风险均衡
- aggressive_growth: 高成长 — 高速成长 / 估值空间 / 赛道催化 / 短期资金博弈

# 默认主映射 (无 user_message 信号时):
| investment_objective | plan_id |
|---|---|
| capital_preservation | capital_preservation |
| stable_growth | stable_growth |
| balanced | balanced |
| aggressive_growth | aggressive_growth |

# Override 例外 (user_message 含特定信号时, 你可偏离默认主映射):
1. user_message 含 "短期机会 / 波段 / 抓机会" + objective != capital_preservation → 可 override 到 aggressive_growth
2. user_message 含 "避险 / 防御 / 担心下跌 / 保本" + objective != capital_preservation → 可 override 到 capital_preservation
3. user_message 含 "长期持有 / 不在乎短期波动 / 长跑" + objective != aggressive_growth → 可 override 到 stable_growth
4. user_message 含 "稳定收益 / 红利 / 股息" → 可 override 到 stable_growth
5. user_message 含 "全面分析 / 综合判断" → balanced

无 override 信号则严格走主映射。

# 任务

客户输入:
- investment_objective: {investment_objective}
- investment_horizon: {investment_horizon}
- risk_tolerance: {risk_tolerance}
- client_total_aum: {client_total_aum}
- client_existing_position: {client_existing_position}
- target_ts_code: {target_ts_code}
- user_message: {user_message}

# 输出

严格输出符合 JSON schema 的对象:
{{
  "plan_id": "<one_of: capital_preservation | stable_growth | balanced | aggressive_growth>",
  "rationale": "<≤200 字符: 为什么选此 plan_id, 是否触发了 override 例外, 哪条>"
}}

# Critic feedback (如有)
{critic_feedback}
"""


def build_router_prompt(state: ResearchState) -> str:
    """Build constrained router system prompt with β override rules."""
    return _PLANNER_SYSTEM_PROMPT_TEMPLATE.format(
        investment_objective=state.investment_objective,
        investment_horizon=state.investment_horizon,
        risk_tolerance=state.risk_tolerance,
        client_total_aum=state.client_total_aum,
        client_existing_position=state.client_existing_position or "未知",
        target_ts_code=state.target_ts_code,
        user_message=state.user_message or "(无)",
        critic_feedback=(
            f"上一轮 plan 被 Critic 评分 < 8.5, 反馈:{state.planner_critic_feedback}"
            if state.planner_critic_feedback
            else "(无 — 第一轮 router)"
        ),
    )


class ResearchPlanner(Agent):
    """v0.8.5: constrained LLM router."""

    def __init__(self, llm: LLMService) -> None:
        self._llm = llm

    def step(self, state: ResearchState) -> StepResult:
        from app.agents.plan_registry import instantiate_plan

        prompt = build_router_prompt(state)
        # LLM 调用 — schema 强约束 plan_id ∈ Literal 4 值
        response = self._llm.chat(
            prompt=prompt, tier="default", schema=ResearchPlan,
        )
        plan = response.parsed  # ResearchPlan(plan_id=..., rationale=...)

        # Hardcoded subtasks instantiate from registry
        subtasks = instantiate_plan(
            plan_id=plan.plan_id,
            target_name=state.target_entity or state.target_ts_code,
            ts_code=state.target_ts_code,
        )
        plan.subtasks = subtasks

        return StepResult(
            state_update={"plan": plan},
            span_metadata={
                "agent": "ResearchPlanner",
                "plan_id": plan.plan_id,
                "retry_count": state.planner_retry_count,
            },
        )
```

```python
# backend/app/agents/critic_subagents/plan_correctness_scorer.py
"""LLM-as-judge plan_correctness scorer (v0.8.5)."""
from __future__ import annotations

from typing import Any

from pydantic import BaseModel, Field

from app.agents.base import Agent
from app.agents.schemas import CriticDimensionScore, ResearchState, StepResult
from app.services.llm_service import LLMService


_PLAN_CORRECTNESS_JUDGE_PROMPT = """你是评测员, 评判 LLM router 选的 plan_id 是否符合规则。

# 评测规则 (v0.8.5 spec β 模式)

主映射:
- objective=capital_preservation → plan_id=capital_preservation
- objective=stable_growth → plan_id=stable_growth
- objective=balanced → plan_id=balanced
- objective=aggressive_growth → plan_id=aggressive_growth

Override 例外:
1. user_message 含 "短期机会/波段" → aggressive_growth
2. user_message 含 "避险/防御/担心下跌" → capital_preservation
3. user_message 含 "长期持有/不在乎波动" → stable_growth
4. user_message 含 "稳定红利/股息" → stable_growth
5. user_message 含 "综合判断" → balanced

# 输入

- investment_objective: {objective}
- user_message: {user_message}
- LLM 选的 plan_id: {plan_id}
- LLM 给的 rationale: {rationale}

# 输出 (0-10 评分 + reasoning)

严格 JSON:
{{
  "score": <float 0-10>,
  "reasoning": "<≤200 字符 评判理由>"
}}

评分标准:
- 10: plan_id 完美符合规则 + rationale 解释充分
- 8.5-9.5: plan_id 符合规则, rationale 有效但简洁
- 6-8.5: plan_id 选对但 rationale 模糊 / 选错且 rationale 无 override 触发解释
- < 6: plan_id 明显错误且 rationale 不能合理解释
"""


class _PlanCorrectnessScore(BaseModel):
    score: float = Field(ge=0.0, le=10.0)
    reasoning: str = Field(max_length=300)


class PlanCorrectnessScorer(Agent):
    """LLM-as-judge that evaluates whether plan_id selection matches β rules."""

    def __init__(self, llm: LLMService) -> None:
        self._llm = llm

    def step(self, state: ResearchState) -> StepResult:  # type: ignore[override]
        plan = state.plan
        if plan is None:
            score = CriticDimensionScore(
                dimension="plan_correctness", score=0.0, evidence="no plan available"
            )
            return StepResult(state_update={"_pc_dim": score})
        prompt = _PLAN_CORRECTNESS_JUDGE_PROMPT.format(
            objective=state.investment_objective,
            user_message=state.user_message or "(无)",
            plan_id=plan.plan_id,
            rationale=plan.rationale,
        )
        resp = self._llm.chat(prompt=prompt, tier="default", schema=_PlanCorrectnessScore)
        parsed: _PlanCorrectnessScore = resp.parsed
        dim = CriticDimensionScore(
            dimension="plan_correctness",
            score=parsed.score,
            evidence=parsed.reasoning,
        )
        return StepResult(state_update={"_pc_dim": dim})
```

```python
# backend/app/orchestration/research_graph.py
from typing import Literal as _L

from langgraph.graph import END

from app.agents.schemas import ResearchState

_PLAN_CORRECTNESS_THRESHOLD = 8.5
_MAX_PLANNER_RETRY = 2


def _retry_router(state: ResearchState) -> _L["retry", "continue"]:
    """Decide whether to retry planner based on critic plan_correctness score."""
    if state.critic_report is None:
        return "continue"
    plan_score = state.critic_report.get_score("plan_correctness")
    if plan_score is None:
        return "continue"
    if plan_score < _PLAN_CORRECTNESS_THRESHOLD and state.planner_retry_count < _MAX_PLANNER_RETRY:
        return "retry"
    return "continue"


def _retry_state_update(state: ResearchState) -> dict[str, Any]:
    """When retry triggers: increment retry_count, stash critic feedback."""
    feedback = ""
    if state.critic_report is not None:
        for dim in state.critic_report.dimensions:
            if dim.dimension == "plan_correctness":
                feedback = dim.evidence
                break
    return {
        "planner_retry_count": state.planner_retry_count + 1,
        "planner_critic_feedback": feedback[:300],  # max_length 字段约束
    }
```
