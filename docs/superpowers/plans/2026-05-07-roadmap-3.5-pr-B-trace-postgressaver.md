"""TraceService — span persistence backed by PostgreSQL via SQLAlchemy.

Roadmap #3.5 PR B — migrated from raw sqlite3 to SQLAlchemy ORM. Public
API preserved: write_span / get_trace / query_spans.
"""

from __future__ import annotations

from typing import Any

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models.trace import Span as SpanORM
from app.services.trace_models import Span, TraceTree


class TraceService:
    """Persists spans for trace inspection.

    Use via dependency injection: pass a SQLAlchemy Session created by the
    standard `get_db` FastAPI dependency.
    """

    def __init__(self, session: Session) -> None:
        self._session = session

    # init_schema() removed — alembic manages schema (#3.5 PR A)

    def write_span(self, span: Span) -> None:
        """INSERT OR REPLACE — uses session.merge for upsert semantics."""
        orm = SpanORM(
            span_id=span.span_id,
            request_id=span.request_id,
            parent_id=span.parent_id,
            name=span.name,
            inputs=span.inputs,
            outputs=span.outputs,
            span_metadata=span.metadata,
            started_at=span.started_at,
            ended_at=span.ended_at,
            error=span.error,
        )
        self._session.merge(orm)
        self._session.commit()

    def get_trace(self, request_id: str) -> TraceTree:
        spans = self.query_spans({"request_id": request_id})
        return TraceTree(spans=spans)

    def query_spans(self, filters: dict[str, Any]) -> list[Span]:
        stmt = select(SpanORM)
        for k, v in filters.items():
            attr_name = "span_metadata" if k == "metadata" else k
            stmt = stmt.where(getattr(SpanORM, attr_name) == v)
        rows = self._session.scalars(stmt).all()
        return [self._orm_to_span(orm) for orm in rows]

    @staticmethod
    def _orm_to_span(orm: SpanORM) -> Span:
        return Span(
            span_id=orm.span_id,
            request_id=orm.request_id,
            parent_id=orm.parent_id,
            name=orm.name,
            inputs=orm.inputs,
            outputs=orm.outputs,
            metadata=orm.span_metadata,
            started_at=orm.started_at,
            ended_at=orm.ended_at,
            error=orm.error,
        )
```

"""LangGraph checkpoint saver factory.

Roadmap #3.5 PR B — migrated from SqliteSaver/AsyncSqliteSaver to
PostgresSaver/AsyncPostgresSaver. Connection from POSTGRES_* env vars
(same as app.core.database).
"""

from __future__ import annotations

import os

import psycopg
from langgraph.checkpoint.postgres import PostgresSaver
from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver


def _build_pg_url() -> str:
    user = os.environ.get("POSTGRES_USER", "postgres")
    password = os.environ.get("POSTGRES_PASSWORD", "postgres123")
    host = os.environ.get("POSTGRES_HOST", "localhost")
    port = os.environ.get("POSTGRES_PORT", "5432")
    db = os.environ.get("POSTGRES_DB", "industry_assistant")
    return f"postgresql://{user}:{password}@{host}:{port}/{db}"


def make_chat_checkpointer() -> PostgresSaver:
    """Sync PostgresSaver — used by chat router (sync graph)."""
    conn = psycopg.connect(_build_pg_url(), autocommit=True)
    saver = PostgresSaver(conn)
    saver.setup()  # CREATE TABLE IF NOT EXISTS langgraph_state etc.
    return saver


async def make_async_chat_checkpointer() -> AsyncPostgresSaver:
    """Async PostgresSaver — used by research router (async graph)."""
    cm = AsyncPostgresSaver.from_conn_string(_build_pg_url())
    saver = await cm.__aenter__()
    await saver.setup()
    return saver
```

(注意:`AsyncPostgresSaver.from_conn_string()` 返回的是一个 async context manager —— 这里 escape it via __aenter__,生命周期由 caller 管。)
