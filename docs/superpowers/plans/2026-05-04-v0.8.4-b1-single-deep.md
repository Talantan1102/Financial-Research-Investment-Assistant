/**
 * Submit B-1 form, returns SSE EventSource for live progress.
 */
export function submitResearch(
  req: ResearchRequest,
  onEvent: (e: SSEResearchEvent) => void,
  onError: (err: Error) => void,
): { abort: () => void } {
  const params = new URLSearchParams()
  Object.entries(req).forEach(([k, v]) => v !== undefined && params.append(k, String(v)))
  
  const es = new EventSource(`/api/v0.5/research/stream?${params}`)  // GET via query for SSE
  // OR: use fetch + ReadableStream (POST + SSE),按现有 chat 实现风格
  es.onmessage = (msg) => {
    try {
      const event: SSEResearchEvent = JSON.parse(msg.data)
      onEvent(event)
    } catch (e) {
      onError(e as Error)
    }
  }
  es.onerror = (e) => onError(new Error('SSE connection error'))
  return { abort: () => es.close() }
}
