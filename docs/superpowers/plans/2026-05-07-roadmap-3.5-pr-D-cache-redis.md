```python
"""TushareCache — Redis-backed API response cache.

Roadmap #3.5 PR D — migrated from sqlite blob to Redis SETEX. Public
async API preserved (get / set) so tushare_service callers don't change.

Key format: tushare:<api_name>:<params_hash>
Value format: pickle.dumps(pd.DataFrame)
TTL: classify_ttl(api_name) 4 groups (financial 365d / anns 24h / daily
1h / default 1h)

Failure mode: Redis unavailable → cache miss + warning log → caller falls
through to real Tushare API. Don't crash on cache layer failure.
"""

from __future__ import annotations

import hashlib
import json
import logging
import pickle
from typing import Any

import pandas as pd
from redis.asyncio import Redis

logger = logging.getLogger(__name__)


# TTL classification — kept identical to legacy (frozensets at module level)
_FINANCIAL_APIS = frozenset({"income", "balancesheet", "cashflow", "fina_indicator",
                              "forecast", "dividend"})
_ANNS_APIS = frozenset({"anns_d", "disclosure_date"})
_DAILY_APIS = frozenset({"daily", "daily_basic", "moneyflow", "stk_holdernumber"})


def classify_ttl(api_name: str) -> int:
    """Return TTL in seconds for the given Tushare API."""
    if api_name in _FINANCIAL_APIS:
        return 365 * 24 * 3600
    if api_name in _ANNS_APIS:
        return 24 * 3600
    if api_name in _DAILY_APIS:
        return 3600
    return 3600  # default 1h


def _build_key(api_name: str, params: dict[str, Any]) -> str:
    """Stable key from api_name + sorted-JSON params."""
    params_str = json.dumps(params, sort_keys=True, default=str)
    params_hash = hashlib.sha256(params_str.encode()).hexdigest()[:16]
    return f"tushare:{api_name}:{params_hash}"


class TushareCache:
    """Redis-backed Tushare API response cache.

    Uses pickle for DataFrame serialization. Thread/event-loop safe via
    redis-py async client. No connection pooling concerns — Redis client
    handles internally.
    """

    def __init__(self, redis: Redis) -> None:
        self._redis = redis

    async def get(self, api_name: str, params: dict[str, Any]) -> pd.DataFrame | None:
        """Return cached DataFrame or None if missing/expired/corrupt."""
        key = _build_key(api_name, params)
        try:
            blob = await self._redis.get(key)
        except Exception as e:
            logger.warning("cache get failed for %s: %s — treating as miss", key, e)
            return None
        if blob is None:
            return None
        try:
            df = pickle.loads(blob)
        except Exception as e:
            logger.warning("cache deserialize failed for %s: %s — treating as miss", key, e)
            return None
        if not isinstance(df, pd.DataFrame):
            logger.warning("cached value for %s is not DataFrame — treating as miss", key)
            return None
        return df

    async def set(
        self, api_name: str, params: dict[str, Any], df: pd.DataFrame, ttl_s: int | None = None
    ) -> None:
        """Cache the DataFrame with TTL. ttl_s=None uses classify_ttl(api_name)."""
        key = _build_key(api_name, params)
        ttl = ttl_s if ttl_s is not None else classify_ttl(api_name)
        try:
            blob = pickle.dumps(df)
            await self._redis.set(key, blob, ex=ttl)
        except Exception as e:
            logger.warning("cache set failed for %s: %s — silently dropping", key, e)
            # 不让 cache 写失败 break 业务调用
```

def _wait_for_redis_ready(host: str, port: int, timeout: int = 30) -> None:
    """Poll Redis via PING until accepting connections."""
    deadline = time.time() + timeout
    last_err: Exception | None = None
    while time.time() < deadline:
        if _is_port_listening(host, port):
            try:
                import redis
                client = redis.Redis(host=host, port=port)
                client.ping()
                return
            except Exception as e:
                last_err = e
        time.sleep(2)
    raise TimeoutError(f"Redis not ready after {timeout}s: {last_err}")


@pytest.fixture(scope="session")
def redis_test_container() -> Iterator[dict[str, object]]:
    """Session-scoped Redis container — mirrors pg_test_container.
    
    若已外部启动(127.0.0.1:6379 listening),复用,session 末**不**清理。
    若没启动,fixture 自动 docker compose up -d redis,session 末 down -v 清理。
    """
    backend_dir = Path(__file__).resolve().parents[1]
    repo_root = backend_dir.parent
    compose_file = repo_root / "docker-compose.yml"
    host = os.environ.get("REDIS_HOST", "127.0.0.1")
    port = int(os.environ.get("REDIS_PORT", "6379"))

    started_by_us = False
    if not _is_port_listening(host, port):
        subprocess.run(
            ["docker", "compose", "-f", str(compose_file), "up", "-d", "redis"],
            cwd=str(repo_root),
            check=True,
        )
        started_by_us = True

    try:
        _wait_for_redis_ready(host, port, timeout=30)
        yield {
            "host": host, "port": port,
            "url": f"redis://{host}:{port}/0",
        }
    finally:
        if started_by_us:
            subprocess.run(
                ["docker", "compose", "-f", str(compose_file), "down", "-v"],
                cwd=str(repo_root),
                check=False,
            )
