**Goal:** Build the ChatPane internal components, EscalationConfirmDialog, full ReportsListPage, and F1-F10 frontend polish on top of the Plan 4a foundation. This plan fills the placeholder pages with content (`<MessageList>` / `<TextMessage>` / `<ToolCallCard>` / `<ResearchReportCard>` / `<StreamingIndicator>` / `<InputArea>` / `<CostMeter>`), wires the EscalationConfirmDialog to `POST /api/v0/chat/escalate`, integrates ECharts for `chart_specs`, lands Playwright e2e for the chat → escalate → research → report flow, and verifies all F1-F10 industry polish criteria.

**Architecture:** Plan 4a already shipped App Shell / TopBar / Sidebar / Layout / Routes / valtio stores (`chatSessionsStore` / `currentChatStore` / `escalationStore`) / `ChatApi` REST client / `useChatSSE` hook / TypeScript types. Plan 4b consumes those — nothing in this plan rebuilds the foundation. Components are React 19 function components with TypeScript; rendering hot-paths use `React.memo` + `useDeferredValue` + `react-window` virtualization. Markdown via `marked` + `highlight.js` syntax highlight; KaTeX math via `react-katex`. Chart specs via `echarts-for-react`. Escalation dialog uses Ant Design `Form` with 4 sub-forms (one per `EscalationPacket` signal class) and inline-edit per-field with `FieldEdit` trace into `escalationStore.user_edits`.

import { memo, useCallback, useMemo, useRef } from 'react';
import { VariableSizeList, type ListChildComponentProps } from 'react-window';
import type { ChatMessage } from '@/types/chat';
import { ResearchReportCard } from './ResearchReportCard';
import { SystemMessage } from './SystemMessage';
import { TextMessage } from './TextMessage';
import { ToolCallCard } from './ToolCallCard';

export interface MessageListProps {
  messages: readonly ChatMessage[];
  height?: number;
}

const ESTIMATE_ROW_HEIGHT = 96;

function MessageRouter({ message }: { message: ChatMessage }) {
  switch (message.message_type) {
    case 'tool_call':
      return <ToolCallCard message={message} />;
    case 'research_report':
      return <ResearchReportCard message={message} />;
    case 'system':
      return <SystemMessage message={message} />;
    case 'text':
    default:
      return <TextMessage message={message} />;
  }
}

import hljs from 'highlight.js';
import { Marked } from 'marked';

/**
 * Marked instance configured with highlight.js. Use `mdInstance.parse(content)`
 * to get HTML that will be inserted via `dangerouslySetInnerHTML`.
 *
 * Security: `marked` v15 escapes raw HTML by default. We rely on that and do
 * NOT enable `unsafe` / `mangle` options. KaTeX runs as a post-pass.
 */
export const mdInstance = new Marked({
  gfm: true,
  breaks: true,
  // marked v15 default sanitizes HTML; explicit hooks for code highlight:
  hooks: {
    postprocess(html: string) {
      return html;
    },
  },
  walkTokens(token) {
    if (token.type === 'code') {
      const code = token as { type: 'code'; lang?: string; text: string };
      const lang = code.lang && hljs.getLanguage(code.lang) ? code.lang : 'plaintext';
      const highlighted = hljs.highlight(code.text, { language: lang, ignoreIllegals: true }).value;
      // Inject highlighted HTML by overwriting the rendered output.
      // Marked exposes this through a custom renderer below.
      (code as unknown as { _hljs?: string })._hljs = `<pre><code class="hljs language-${lang}">${highlighted}</code></pre>`;
    }
  },
});

mdInstance.use({
  renderer: {
    code(token) {
      const t = token as { _hljs?: string };
      if (t._hljs) return t._hljs;
      return false; // fall through to default
    },
  },
});

export function renderMarkdown(content: string): string {
  return mdInstance.parse(content) as string;
}

import katex from 'katex';
// ... (existing imports)

const INLINE_MATH = /\$([^\n$]+?)\$/g;
const BLOCK_MATH = /\$\$([\s\S]+?)\$\$/g;

function renderKatex(html: string): string {
  // Block first to avoid greedy inline match.
  let out = html.replace(BLOCK_MATH, (_, expr: string) => {
    try {
      return katex.renderToString(expr, { displayMode: true, throwOnError: false });
    } catch {
      return _;
    }
  });
  out = out.replace(INLINE_MATH, (_, expr: string) => {
    try {
      return katex.renderToString(expr, { displayMode: false, throwOnError: false });
    } catch {
      return _;
    }
  });
  return out;
}

export function renderMarkdown(content: string): string {
  const raw = mdInstance.parse(content) as string;
  return renderKatex(raw);
}

import { SendOutlined } from '@ant-design/icons';
import { Button } from 'antd';
import { useCallback, useEffect, useRef, useState, type KeyboardEvent } from 'react';
import { useSnapshot } from 'valtio';
import { currentChatStore } from '@/store/currentChatStore';
import styles from '@/styles/chat.module.scss';

export interface InputAreaProps {
  sessionId?: string;
  onSend?: (text: string) => void;
  onAbort?: () => void;
  onEscalate?: () => void;
}

const MIN_ROWS_HEIGHT = 44; // px
const MAX_ROWS_HEIGHT = 240; // px

export function InputArea(props: InputAreaProps) {
  const [value, setValue] = useState('');
  const taRef = useRef<HTMLTextAreaElement>(null);
  const snap = useSnapshot(currentChatStore);
  const streaming = snap.streaming_phase !== 'idle';

  const autoResize = useCallback(() => {
    const ta = taRef.current;
    if (!ta) return;
    ta.style.height = 'auto';
    const next = Math.min(Math.max(ta.scrollHeight, MIN_ROWS_HEIGHT), MAX_ROWS_HEIGHT);
    ta.style.height = `${next}px`;
  }, []);

  useEffect(() => { autoResize(); }, [value, autoResize]);

  const send = useCallback(() => {
    const text = value.trim();
    if (!text) return;
    props.onSend?.(text);
    setValue('');
  }, [value, props]);

  const onKey = useCallback((e: KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === 'Enter' && !e.shiftKey && !e.nativeEvent.isComposing) {
      e.preventDefault();
      send();
    }
  }, [send]);

  return (
    <div className={styles.inputArea} data-session={props.sessionId ?? ''}>
      <textarea
        ref={taRef}
        data-testid="input-textarea"
        className={styles.inputTextarea}
        value={value}
        onChange={(e) => setValue(e.target.value)}
        onKeyDown={onKey}
        placeholder={streaming ? '正在生成中...' : '问点什么 (Enter 发送, Shift+Enter 换行)'}
        rows={1}
      />
      <div className={styles.inputActions}>
        <Button type="primary" icon={<SendOutlined />} onClick={send} disabled={!value.trim() || streaming} aria-label="发送">发送</Button>
      </div>
    </div>
  );
}

```ts
import { useCallback, useEffect, useState } from 'react';

export interface UseScrollStickResult {
  isAtBottom: boolean;
  scrollToBottom: () => void;
}

/**
 * IntersectionObserver-based auto-stick — observes a sentinel `<div>` placed
 * at the bottom of the messages region. While the sentinel is visible,
 * `isAtBottom=true` and the caller may auto-scroll on new messages. As soon
 * as the user scrolls up (sentinel exits viewport), `isAtBottom=false` and
 * we stop sticking — respecting user scrollback.
 */
export function useScrollStick(sentinel: Element | null): UseScrollStickResult {
  const [isAtBottom, setIsAtBottom] = useState(true);

  useEffect(() => {
    if (!sentinel) return;
    const obs = new IntersectionObserver(
      (entries) => setIsAtBottom(entries[0]?.isIntersecting ?? false),
      { threshold: 0.01 },
    );
    obs.observe(sentinel);
    return () => obs.disconnect();
  }, [sentinel]);

  const scrollToBottom = useCallback(() => {
    sentinel?.scrollIntoView({ behavior: 'smooth', block: 'end' });
  }, [sentinel]);

  return { isAtBottom, scrollToBottom };
}
```

import type { EscalationPacket, FieldEdit } from '@/types/escalation';

interface EscalationState {
  dialog_open: boolean;
  session_id: string | null;
  packet_draft: EscalationPacket | null;
  user_edits: FieldEdit[];
  submitting: boolean;
  error: string | null;
}

export const escalationStore: EscalationState = proxy({
  dialog_open: false,
  session_id: null,
  packet_draft: null,
  user_edits: [],
  submitting: false,
  error: null,
});

export function recordUserEdit(edit: FieldEdit): void {

export function InlineEditField({ fieldPath, llmValue, multiline }: InlineEditFieldProps) {
  const initial = asDisplay(llmValue);

const CHART_FENCE = /```chart_spec\n([\s\S]+?)\n```/g;

function extractCharts(content: string): { stripped: string; charts: Array<{ id: string; spec: ChartSpec }> } {
  const charts: Array<{ id: string; spec: ChartSpec }> = [];
  let i = 0;
  const stripped = content.replace(CHART_FENCE, (_, json: string) => {
    try {
      const spec = JSON.parse(json) as ChartSpec;
      const id = `chart-${i++}`;
      charts.push({ id, spec });
      return `<div data-chart-spec-id="${id}"></div>`;
    } catch {
      return '`(chart_spec parse error)`';
    }
  });
  return { stripped, charts };
}

export interface MarkdownRenderResult { html: string; charts: Array<{ id: string; spec: ChartSpec }>; }

export function renderMarkdownWithCharts(content: string): MarkdownRenderResult {
  const { stripped, charts } = extractCharts(content);
  const raw = mdInstance.parse(stripped) as string;
  const html = renderKatex(raw);
  return { html, charts };
}
