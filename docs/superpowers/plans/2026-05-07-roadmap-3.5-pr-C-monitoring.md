# Roadmap #3.5 PR C — Monitoring SQLite → PostgreSQL Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Migrate B-3 持仓预警 monitoring system 5 sqlite tables(monitoring_customers / monitoring_runs / monitoring_signals / monitoring_alerts / notifications)→ PostgreSQL via SQLAlchemy ORM。监控服务的 5 个 SignalRule 评分 + escalation + email + APScheduler 行为不变。

**Architecture:** 当前 monitoring 17 个文件,1481 LOC,**完全用 raw sqlite3** —— 没有 ORM 层。本 PR 加一层 ORM(`backend/app/models/monitoring.py` 5 个 model),把 monitoring_service.py 的 `_record_run` / `_record_signal` / `_record_alert` / `_send_notifications` / `_load_customer` 等 5 个持久化方法 + monitoring_router.py 的 9 个 CRUD 端点全部改成 SQLAlchemy session 模式。`init_monitoring_tables.py` 删掉(alembic 接管)。

**Tech Stack:** SQLAlchemy 2.0,PostgreSQL 15(JSONB for thresholds_override / raw_data_ref / report_json),保持现有 APScheduler / EmailNotifier / EscalationCoordinator 接口不变。

**Spec:** `docs/superpowers/specs/2026-05-07-roadmap-3.5-db-unify-design.md` § 4.2(monitoring 系列表)

**Prerequisites:** PR A merged。可以并行于 PR B。

```python
"""Roadmap #3.5 PR C — Monitoring B-3 ORM models.

Mirrors legacy sqlite schema from app/scripts/init_monitoring_tables.py.
PG-native: JSONB for JSON cols, TIMESTAMP for timestamps, FK constraints
explicit.
"""

from __future__ import annotations

from datetime import datetime
from typing import TYPE_CHECKING

from sqlalchemy import Boolean, Column, DateTime, ForeignKey, Integer, JSON, Numeric, String, Text
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import relationship

from app.core.database import Base


class MonitoringCustomer(Base):
    __tablename__ = "monitoring_customers"
    
    id = Column(String(64), primary_key=True)
    ts_code = Column(String(32), nullable=False, unique=True)
    name = Column(String(128), nullable=True)
    industry = Column(String(64), nullable=True)
    enabled = Column(Boolean, nullable=False, default=True)
    thresholds_override = Column(JSONB().with_variant(JSON(), "sqlite"), nullable=True)
    created_at = Column(DateTime(timezone=False), nullable=False, default=datetime.utcnow)
    updated_at = Column(DateTime(timezone=False), nullable=False, default=datetime.utcnow, onupdate=datetime.utcnow)


class MonitoringRun(Base):
    __tablename__ = "monitoring_runs"
    
    id = Column(String(64), primary_key=True)
    customer_id = Column(String(64), ForeignKey("monitoring_customers.id"), nullable=False, index=True)
    trigger_type = Column(String(32), nullable=False)  # cron / disclosure_event / manual
    started_at = Column(DateTime(timezone=False), nullable=False)
    finished_at = Column(DateTime(timezone=False), nullable=True)
    status = Column(String(32), nullable=False)  # success / failed
    error_message = Column(Text, nullable=True)
    cost_cny = Column(Numeric(10, 4), nullable=False, default=0)


class MonitoringSignal(Base):
    __tablename__ = "monitoring_signals"
    
    id = Column(String(64), primary_key=True)
    run_id = Column(String(64), ForeignKey("monitoring_runs.id"), nullable=False, index=True)
    rule_name = Column(String(64), nullable=False)
    level = Column(String(16), nullable=False)  # green/yellow/red
    detected_value = Column(Numeric(12, 4), nullable=True)
    threshold = Column(Numeric(12, 4), nullable=True)
    explanation = Column(Text, nullable=True)
    raw_data_ref = Column(JSONB().with_variant(JSON(), "sqlite"), nullable=True)


class MonitoringAlert(Base):
    __tablename__ = "monitoring_alerts"
    
    id = Column(String(64), primary_key=True)
    run_id = Column(String(64), ForeignKey("monitoring_runs.id"), nullable=False, index=True)
    customer_id = Column(String(64), ForeignKey("monitoring_customers.id"), nullable=False, index=True)
    alert_level = Column(String(16), nullable=False)
    report_json = Column(JSONB().with_variant(JSON(), "sqlite"), nullable=False)
    report_markdown = Column(Text, nullable=True)
    escalated = Column(Boolean, nullable=False, default=False)
    escalation_status = Column(String(64), nullable=True)
    deep_dive_text = Column(Text, nullable=True)
    created_at = Column(DateTime(timezone=False), nullable=False, default=datetime.utcnow)


class Notification(Base):
    __tablename__ = "notifications"
    
    id = Column(String(64), primary_key=True)
    alert_id = Column(String(64), ForeignKey("monitoring_alerts.id"), nullable=False, index=True)
    channel = Column(String(32), nullable=False)  # in_app / email
    recipient = Column(String(128), nullable=True)
    sent_at = Column(DateTime(timezone=False), nullable=True)
    send_status = Column(String(32), nullable=False)
    error_message = Column(Text, nullable=True)
    read_at = Column(DateTime(timezone=False), nullable=True)
```

- [ ] **Step 2: Replace `_connect()` helper with session injection**

OLD pattern:
```python

```

NEW: 删掉 `_connect`,monitoring_service `__init__` 接受 `session_factory: Callable[[], Session]` 或 直接 `session: Session` 参数。考虑到 `run_scan` 是 background task,**session_factory 模式**(每个请求拿新 session)更合适。

- [ ] **Step 3: Rewrite each persist method**

`_record_run(customer_id, trigger_type)`:

```python
def _record_run(self, customer_id: str, trigger_type: str) -> str:
    run_id = str(uuid4())
    with self._session_factory() as session:
        run = MonitoringRun(
            id=run_id, customer_id=customer_id, trigger_type=trigger_type,
            started_at=datetime.utcnow(), status="started",
        )
        session.add(run)
        session.commit()
    return run_id
```

`_record_signal(run_id, rule_name, level, detected_value, threshold, explanation, raw_data_ref)`:

```python
def _record_signal(self, run_id: str, rule_name: str, level: SignalLevel, ...) -> None:
    with self._session_factory() as session:
        sig = MonitoringSignal(
            id=str(uuid4()), run_id=run_id, rule_name=rule_name,
            level=level.value, detected_value=detected_value,
            threshold=threshold, explanation=explanation, raw_data_ref=raw_data_ref,
        )
        session.add(sig)
        session.commit()
```

`_record_alert(...)`、`_send_notifications(...)`、`_load_customer(customer_id)` 同样模式重写。

- [ ] **Step 4: Update `run_scan` 主流程**

```python
async def run_scan(self, customer_ids: list[str], *, trigger_type: TriggerType = "manual") -> list[str]:
    run_ids = []
    for customer_id in customer_ids:
        customer = self._load_customer(customer_id)
        if not customer or not customer.enabled:
            continue
        run_id = self._record_run(customer_id, trigger_type)
        try:
            level, signals = await self._detector.detect(customer, ...)
            for sig in signals:
                self._record_signal(run_id, sig.rule_name, sig.level, ...)
            if level == SignalLevel.RED:
                # escalation pipeline ...
                self._record_alert(run_id, customer_id, level, ...)
            self._mark_run_finished(run_id, status="success")
        except Exception as e:
            self._mark_run_finished(run_id, status="failed", error_message=str(e))
        run_ids.append(run_id)
    return run_ids
```

- [ ] **Step 2: Replace each endpoint's sqlite3 → SQLAlchemy session**

每个 endpoint 用 `Depends(get_db)` 拿 session,把 `cursor.execute(...)` / `cursor.fetchall()` / `cursor.fetchone()` 换成 SQLAlchemy `session.query(...).filter(...)` / `session.add(...)` / `session.commit()`.

例子(GET /customers):

```python
@router.get("/customers", response_model=list[CustomerOut])
def list_customers(session: Session = Depends(get_db)) -> list[CustomerOut]:
    customers = session.query(MonitoringCustomer).all()
    return [CustomerOut.from_orm(c) for c in customers]
```

例子(GET /alerts/{aid} — LEFT JOIN notifications):

```python
@router.get("/alerts/{aid}", response_model=AlertDetailOut)
def get_alert_detail(aid: str, session: Session = Depends(get_db)) -> AlertDetailOut:
    alert = session.query(MonitoringAlert).filter(MonitoringAlert.id == aid).first()
    if not alert:
        raise HTTPException(status_code=404)
    notifications = session.query(Notification).filter(Notification.alert_id == aid).all()
    return AlertDetailOut(alert=alert, notifications=notifications)
```

(其他 7 endpoint 类似 pattern。)
