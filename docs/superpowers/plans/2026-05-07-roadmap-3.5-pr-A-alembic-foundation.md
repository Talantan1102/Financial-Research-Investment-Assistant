"""Roadmap #3.5 PR A — verify all UUID columns use the standard with_variant
pattern so PG and sqlite tests both work, and FKs don't fail with type mismatch.

Asserts on schema definition, not data — pure unit test.
"""

from __future__ import annotations

import pytest
from sqlalchemy import String
from sqlalchemy.dialects.postgresql import UUID as PG_UUID

from app.models.chat import ChatAttachment, ChatMessage, ChatSession, LongTermMemory
from app.models.knowledge import Document, KnowledgeBase
from app.models.research_report import ResearchReport
from app.models.user import User


def _column_is_uuid_with_sqlite_variant(column) -> bool:
    """A UUID-typed column with sqlite variant returning String(36).

    Returns True if column.type compiles to UUID on PG and String(36) on sqlite.
    """
    col_type = column.type
    # On PG, type should be PG_UUID
    if not isinstance(col_type, PG_UUID):
        return False
    # Check sqlite variant exists and is String
    variant = col_type.dialect_impl.cache.get("sqlite") if hasattr(col_type, "dialect_impl") else None
    # Simpler check: SQLAlchemy stores variants in a dict
    variants = getattr(col_type, "_variant_mapping", {}) or getattr(col_type, "variants", {})
    sqlite_variant = variants.get("sqlite")
    if sqlite_variant is None:
        return False
    return isinstance(sqlite_variant, String) and sqlite_variant.length == 36


@pytest.mark.parametrize(
    "model,column_name",
    [
        (User, "id"),
        (ChatSession, "user_id"),
        (ChatMessage, "user_id"),
        (ChatAttachment, "user_id"),
        (LongTermMemory, "user_id"),
        (KnowledgeBase, "user_id"),
        (Document, "user_id"),
        (ResearchReport, "user_id"),
    ],
)
def test_uuid_column_uses_sqlite_variant(model, column_name):
    """Every UUID column (PK or FK to users) must use PG_UUID + with_variant(String(36), 'sqlite').

    Why: sqlite-override tests need String for portability, PG needs UUID for FK type
    compatibility. Without this, Base.metadata.create_all() fails on a fresh PG with
    DatatypeMismatch on FK constraints (PR #21 root cause).
    """
    col = model.__table__.columns[column_name]
    assert _column_is_uuid_with_sqlite_variant(col), (
        f"{model.__name__}.{column_name} type={col.type!r} does not use "
        f"PG_UUID(as_uuid=True).with_variant(String(36), 'sqlite'). "
        f"See backend/app/models/research_report.py:49 for reference pattern."
    )
```

"""Alembic environment script.

Reads PG connection from POSTGRES_* env vars (same source as app), imports
all SQLAlchemy models so autogenerate sees them.

Roadmap #3.5 PR A — single source schema management.
"""

from __future__ import annotations

import os
from logging.config import fileConfig

from alembic import context
from sqlalchemy import engine_from_config, pool

# Alembic Config object — provides access to alembic.ini values.
config = context.config

# Configure logging from alembic.ini.
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# Import all models barrel so Base.metadata is populated.
# Without this, autogenerate misses every table and writes an empty migration.
from app.core.database import Base  # noqa: E402
from app import models  # noqa: E402, F401  — barrel import side effect

target_metadata = Base.metadata


def _build_db_url() -> str:
    """Read POSTGRES_* env vars (same as app/core/database.py)."""
    user = os.environ.get("POSTGRES_USER", "postgres")
    password = os.environ.get("POSTGRES_PASSWORD", "postgres123")
    host = os.environ.get("POSTGRES_HOST", "localhost")
    port = os.environ.get("POSTGRES_PORT", "5432")
    db = os.environ.get("POSTGRES_DB", "industry_assistant")
    return f"postgresql://{user}:{password}@{host}:{port}/{db}"


def run_migrations_offline() -> None:
    """Run migrations in offline mode (generates SQL without DB connection)."""
    context.configure(
        url=_build_db_url(),
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in online mode (real DB connection)."""
    config_section = config.get_section(config.config_ini_section, {})
    config_section["sqlalchemy.url"] = _build_db_url()
    connectable = engine_from_config(
        config_section,
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
        )
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```
